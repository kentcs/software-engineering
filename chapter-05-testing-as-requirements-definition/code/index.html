from behave import given, when, then
from selenium import webdriver
from selenium.webdriver.common.by import By
from selenium.webdriver.common.keys import Keys
from selenium.webdriver.support.ui import WebDriverWait
from selenium.webdriver.support import expected_conditions as EC


@given("I have opened the Wikipedia home page")
def open_home_page(context):
    # Set up the precondition...
    context.browser = webdriver.Chrome()
    context.add_cleanup(context.browser.quit)
    context.wait = WebDriverWait(context.browser, 15)
    context.browser.get("https://en.wikipedia.org/wiki/Main_Page")
    # ...and verify it: the home page is up and its search box is usable.
    context.wait.until(EC.element_to_be_clickable((By.NAME, "search")))


@when("I search for Light Bulb")
def search_for_light_bulb(context):
    start = context.browser.current_url
    box = context.browser.find_element(By.NAME, "search")
    box.send_keys("Light Bulb", Keys.ENTER)
    # Verify the action was taken: we left the home page for an article.
    context.wait.until(EC.url_changes(start))
    heading = context.browser.find_element(By.ID, "firstHeading").text
    assert not heading.startswith("Search results"), heading


@then("the page text contains Hanks")
def page_text_contains_edison(context):
    text = context.browser.find_element(By.TAG_NAME, "body").text
    assert "Edisonn" in text

06-hooks/features/environment.py

import os
import re

from selenium import webdriver
from selenium.webdriver.support.ui import WebDriverWait


def before_all(context):
    # Once per run: start one browser for every scenario to share.
    context.browser = webdriver.Chrome()
    context.wait = WebDriverWait(context.browser, 15)


def before_scenario(context, scenario):
    # Once per scenario: reset what a scenario could leave behind.
    context.browser.delete_all_cookies()


def after_scenario(context, scenario):
    # Teardown can look at the outcome. Keep a picture of failures.
    if scenario.status == "failed":
        name = re.sub(r"\W+", "-", scenario.name).strip("-").lower()
        os.makedirs("screenshots", exist_ok=True)
        context.browser.save_screenshot(f"screenshots/{name}.png")


def after_all(context):
    context.browser.quit()

07-spreadsheet-parser/features/cells.py

"""Turn the text typed into a spreadsheet cell into a number."""


def parse(text):
    sign = 1.0
    if text.startswith(("-", "+")):
        if text[0] == "-":
            sign = -1.0
        text = text[1:]
    parts = text.split(".")
    if (len(parts) > 2 or
            not any(parts) or
            any(c < "0" or c > "9" for part in parts for c in part)):
        raise ValueError("Expected digits with an optional decimal point")
    value = 0.0
    place = 0.1
    after_point = False
    for digit in text:
        if digit == ".":
            after_point = True
        elif after_point:
            value += (ord(digit) - ord("0")) * place
            place /= 10
        else:
            value = value * 10 + ord(digit) - ord("0")
    return sign * value

07-spreadsheet-parser/features/numbers_in_cells.feature

Feature: Numbers typed into spreadsheet cells
  My spreadsheet lives or dies on reading what I type the way I mean it.
  A cell that holds a number must hold exactly that number, and a cell
  that does not hold a number must say so instead of guessing.

  Scenario Outline: A cell that holds a number
    When I type "<entry>" into a cell
    Then the cell holds the number <number>

    Examples: Whole numbers
      | entry | number |
      | 7     | 7      |
      | 0     | 0      |
      | 42    | 42     |
      | 105   | 105    |
      | 007   | 7      |

    Examples: Decimals
      | entry | number |
      | 12.5  | 12.5   |
      | 0.25  | 0.25   |
      | 10.02 | 10.02  |
      | .5    | 0.5    |
      | 5.    | 5      |

    Examples: Signs
      | entry | number |
      | -12.5 | -12.5  |
      | +7    | 7      |
      | -0.25 | -0.25  |

  Scenario Outline: A cell that does not hold a number
    When I type "<entry>" into a cell
    Then the cell is rejected as not a number

    Examples: Not numbers
      | entry |
      | abc   |
      | 12x   |
      | 1.2.3 |
      | +     |
      | -     |
      | .     |
      | --7   |
      | 1e3   |

07-spreadsheet-parser/features/steps/cell_steps.py

import math

from behave import when, then

from cells import parse


@when('I type "{entry}" into a cell')
def type_into_cell(context, entry):
    # Do the action and keep what happened: a value or an error.
    context.entry = entry
    context.value = None
    context.error = None
    try:
        context.value = parse(entry)
    except ValueError as error:
        context.error = error


@then("the cell holds the number {number}")
def cell_holds_number(context, number):
    assert context.error is None, f"{context.entry!r} was rejected"
    expected = float(number)
    assert math.isclose(context.value, expected, rel_tol=1e-12, abs_tol=1e-12), \
        f"{context.entry!r} gave {context.value!r}, expected {expected!r}"


@then("the cell is rejected as not a number")
def cell_is_rejected(context):
    assert context.error is not None, \
        f"{context.entry!r} was accepted as {context.value!r}"

test_light_bulb_search.py

from selenium import webdriver
from selenium.webdriver.common.by import By
from selenium.webdriver.common.keys import Keys
from selenium.webdriver.support.ui import WebDriverWait
from selenium.webdriver.support import expected_conditions as EC


def test_light_bulb_mentions_edison():
    browser = webdriver.Chrome()
    try:
        browser.get("https://en.wikipedia.org/wiki/Main_Page")
        wait = WebDriverWait(browser, 15)
        search = wait.until(EC.element_to_be_clickable((By.NAME, "search")))
        start = browser.current_url
        search.send_keys("light bulb", Keys.ENTER)
        wait.until(EC.url_changes(start))
        wait.until(EC.visibility_of_element_located((By.ID, "firstHeading")))
        text = browser.find_element(By.TAG_NAME, "body").text
        assert "Edison" in text
    finally:
        browser.quit()


def main():
    test_light_bulb_mentions_edison()


if __name__ == "__main__":
    main()

"""Fetch the text of a Wikipedia page."""
import requests

HEADERS = {"User-Agent": "SoftwareEngineeringCourse/1.0 (educational example)"}


def page_text(title):
    url = "https://en.wikipedia.org/wiki/" + title
    response = requests.get(url, headers=HEADERS, timeout=15)
    response.raise_for_status()
    return response.text
